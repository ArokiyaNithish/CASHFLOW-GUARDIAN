CREATE TABLE companies (

    user_id      UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    company_id   UUID REFERENCES companies(company_id),
    name         TEXT NOT NULL,
    email        TEXT UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    role         TEXT CHECK (role IN ('owner','finance_manager','viewer')) DEFAULT 'viewer',
    created_at   TIMESTAMPTZ DEFAULT now()
);
-- ============ FINANCIAL DATA ============
CREATE TABLE customers (
    customer_id   UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    company_id    UUID REFERENCES companies(company_id),
    name          TEXT NOT NULL,
    avg_delay_days NUMERIC(5,2) DEFAULT 0,     -- rolling avg, updated by ML job
    reliability_score NUMERIC(5,2) DEFAULT 100 -- 0-100, updated by ML job
);
CREATE TABLE suppliers (
    supplier_id   UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    company_id    UUID REFERENCES companies(company_id),
    name          TEXT NOT NULL,
    negotiability TEXT CHECK (negotiability IN ('high','medium','low')) DEFAULT 'medium',
    penalty_rate  NUMERIC(5,2) DEFAULT 0        -- % penalty for late payment
);
CREATE TABLE invoices (   -- receivables
    invoice_id    UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    company_id    UUID REFERENCES companies(company_id),
    customer_id   UUID REFERENCES customers(customer_id),
    amount        NUMERIC(14,2) NOT NULL,
    issue_date    DATE NOT NULL,
    due_date      DATE NOT NULL,
    status        TEXT CHECK (status IN ('open','paid','overdue','financed')) DEFAULT 'open',
    predicted_pay_date  DATE,        -- filled by ML
    predicted_pay_prob  NUMERIC(5,2) -- filled by ML, 0-1
);
CREATE TABLE payables (   -- supplier obligations
    payable_id    UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    company_id    UUID REFERENCES companies(company_id),
    supplier_id   UUID REFERENCES suppliers(supplier_id),
    amount        NUMERIC(14,2) NOT NULL,
    due_date      DATE NOT NULL,
    status        TEXT CHECK (status IN ('open','paid','delayed')) DEFAULT 'open',
    priority      TEXT CHECK (priority IN ('critical','normal','flexible')) DEFAULT 'normal'
);
CREATE TABLE recurring_obligations (  -- payroll, rent, EMI, tax, subscriptions
    obligation_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    company_id    UUID REFERENCES companies(company_id),
    type          TEXT CHECK (type IN ('payroll','rent','emi','tax','subscription','utility')),
    amount        NUMERIC(14,2) NOT NULL,
    due_day_of_month INT,           -- e.g. 28 for payroll
    frequency     TEXT CHECK (frequency IN ('monthly','quarterly','annual')) DEFAULT 'monthly'
);
CREATE TABLE transactions (   -- historical bank transactions (synthetic)
    txn_id        UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    company_id    UUID REFERENCES companies(company_id),
    txn_date      DATE NOT NULL,
    amount        NUMERIC(14,2) NOT NULL,   -- negative = outflow
    category      TEXT,                     -- 'receivable','payable','payroll','emi','tax','misc'
    counterparty  TEXT,
    balance_after NUMERIC(14,2)
);
-- ============ AI / AGENT LAYER ============
CREATE TABLE forecasts (
    forecast_id   UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    risk_score    NUMERIC(5,2)   -- 0-100
);
CREATE TABLE risk_events (
    event_id      UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    forecast_id   UUID REFERENCES forecasts(forecast_id),

    reference_id  UUID,          -- invoice_id / payable_id / obligation_id

CREATE TABLE agent_plans (
    plan_id       UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    company_id    UUID REFERENCES companies(company_id),

    options        JSONB,         -- [{label:'A', desc, impact, cost, risk}, ...]

    status         TEXT CHECK (status IN ('proposed','approved','rejected','modified')) DEFAULT 'proposed',
    created_at     TIMESTAMPTZ DEFAULT now()
);
CREATE TABLE agent_actions (
    action_id     UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    plan_id       UUID REFERENCES agent_plans(plan_id),

    target_ref_id UUID,          -- invoice_id / payable_id
    permission_level TEXT CHECK (permission_level IN ('L1_auto','L2_approval','L3_strict')),
    status        TEXT CHECK (status IN ('pending_approval','approved','executed','rejected')) DEFAULT 'pending_approval',
    payload       JSONB,         -- generated content: email draft, negotiation terms

Generation logic (Python, faker + numpy):
Generate customers with a payment-behavior profile: reliable (0-3 day delay), moderate (3-8 day delay), chronic_late (8-20 day delay), each with a std-dev for realism.
Generate suppliers with negotiability and penalty_rate.
Generate recurring obligations (payroll on a fixed day, rent, EMI, quarterly tax).
Generate 6-12 months of transaction history following these profiles, including weekday/weekend and salary-cycle effects.
Deliberately inject one crisis window near "today" in the data: a large customer invoice with a chronic_late customer + a supplier payment + payroll landing within the same 10-15 day window. This is what your forecast/agent will "discover" live in the demo ΓÇö script it, don't leave it to chance.
# generate_dataset.py (skeleton)
import numpy as np, pandas as pd
from faker import Faker
fake = Faker("en_IN")
PROFILES = {
    "reliable":     {"mean_delay": 1,  "std": 1},
    "moderate":     {"mean_delay": 5,  "std": 2},
    "chronic_late": {"mean_delay": 12, "std": 4},
}
def generate_customers(n=50):
    rows = []
    for i in range(n):
        profile = np.random.choice(list(PROFILES), p=[0.5, 0.35, 0.15])
        rows.append({
            "customer_id": f"CUST{i:04d}",
            "name": fake.company(),
            "profile": profile,
            **PROFILES[profile]
        })
    return pd.DataFrame(rows)

5. RAG Corpus & Retrieval
Documents to include (keep small and curated ΓÇö 15-25 short docs, not a huge corpus):
MSME delayed-payment guidance (public/government summaries)
TReDS (Trade Receivables Discounting System) eligibility basics
Generic invoice-financing eligibility criteria
Company-specific policy stubs you write yourself: "Safety reserve policy", "Supplier negotiation policy", "Approval threshold policy"
Stack: LangChain or LlamaIndex + FAISS (simplest, no external service) or Chroma. Chunk size ~300-500 tokens, top-k=3 retrieval.
Usage pattern: RAG is called as a tool by the agent, not blended into every response. Example tool call the agent makes mid-reasoning:
retrieve_policy(query="Is a Γé╣3L overdue invoice eligible for invoice financing?")
ΓåÆ returns 2-3 relevant policy chunks
ΓåÆ agent cites them when constructing Option C (invoice financing)
This is what justifies RAG being a real component rather than decoration ΓÇö call it out explicitly in the demo narration.
